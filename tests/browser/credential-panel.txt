import {definePanel,serverAction,credentials,z} from 'paneon/sdk';
import {useState} from 'react';
import {writeFileSync} from 'node:fs';
import {Button,Page,Stack} from 'paneon/ui';
export const protectedAction=serverAction({credential:{label:'Local credential test',ttlSeconds:900},input:z.object({credential:z.string(),leak:z.boolean()}),handler:async(ctx,input)=>ctx.secrets.use(input.credential,async value=>input.leak?{value}:{length:value.length,newline:value.includes('\n'),quote:value.includes('"')})});
export const shortAction=serverAction({credential:{label:'Short credential test',ttlSeconds:1},input:z.object({credential:z.string(),leak:z.boolean(),hold:z.boolean()}),handler:async(ctx,input)=>ctx.secrets.use(input.credential,async value=>{if(input.hold){writeFileSync('in-use.txt','entered');await new Promise(resolve=>setTimeout(resolve,3000));}return input.leak?{value}:{length:value.length};})});
export const otherAction=serverAction({input:z.object({credential:z.string()}),handler:async(ctx,input)=>ctx.secrets.use(input.credential,async value=>({length:value.length}))});
function Panel(){
 const[ref,setRef]=useState('');const[shortRef,setShortRef]=useState('');const[result,setResult]=useState('');
 async function perform(callback:()=>Promise<unknown>){try{setResult(JSON.stringify(await callback()));}catch(cause){const error=cause as Error&{code?:string};setResult(`${error.code}|${error.message}`);}}
 return <Page title="Credential runtime test"><Stack>
 <Button onClick={()=>void perform(async()=>{const credential=await credentials.request(protectedAction);setRef(credential.ref);return{ready:true};})}>Request credential</Button>
 <Button onClick={()=>void perform(()=>protectedAction({credential:ref,leak:false}))}>Use credential</Button>
 <Button onClick={()=>void perform(()=>protectedAction({credential:ref,leak:true}))}>Leak credential</Button>
 <Button onClick={()=>void perform(()=>otherAction({credential:ref}))}>Wrong action</Button>
 <Button onClick={()=>void perform(async()=>{const credential=await credentials.request(shortAction);setShortRef(credential.ref);return{ready:true};})}>Request short credential</Button>
 <Button onClick={()=>void perform(()=>shortAction({credential:shortRef,leak:false,hold:false}))}>Use short credential</Button>
 <output aria-label="Credential reference">{ref}</output><output aria-label="Short credential reference">{shortRef}</output><pre aria-label="Action result">{result}</pre>
 </Stack></Page>;
}
export default definePanel({id:'credential-test',requires:{paneon:'0.0.0'},input:z.object({}),events:{feedback:z.object({text:z.string()})},render:Panel});
