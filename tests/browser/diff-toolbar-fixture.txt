import { FileDiffNotice } from './shared/file-diff-notice';
import { useMemo, useRef, useState } from 'react';
import { definePanel, z } from '@qoggy/paneon/sdk';
import { Button, usePageAnnotations } from '@qoggy/paneon/ui';
import { DiffView } from './shared/diff.tsx';
import type { DiffViewHandle } from './shared/diff.tsx';
import { createDiffModel } from './shared/diff-model.ts';
import { diffAnnotationTargets, diffTargetLocation, diffTargetText } from './shared/diff-schema.ts';
import type { GitRef } from './shared/diff-schema.ts';

const original = Array.from(
  { length: 90 },
  (_, i) => `const line${i + 1} = "original ${i + 1}";\r\n`,
).join('');
const modified = original
  .replace('original 10', 'changed 10')
  .replace('original 45', 'changed 45')
  .replace('original 80', 'changed 80');
function Panel() {
  const diff = useRef<DiffViewHandle>(null);
  const [kind, setKind] = useState('changed');
  const model = useMemo(
    () => createDiffModel(original, kind === 'same' ? original : modified),
    [kind],
  );
  usePageAnnotations<GitRef>({
    onSubmit: async () => {},
    adapter: {
      resolve: (target) => ({ status: 'ready', text: diffTargetText(model, target) }),
      targetRanges: (target) =>
        target.ref.sourceId === kind
          ? (diff.current?.getRanges(diffTargetLocation(model, target)) ?? [])
          : [],
      locate: (target) => diff.current?.reveal(diffTargetLocation(model, target)),
    },
  });
  return (
    <>
      <div className="flex flex-wrap items-center gap-2 p-2">
        <Button onClick={() => setKind('changed')}>Changed source</Button>
        <Button onClick={() => setKind('same')}>Same source</Button>
        <Button onClick={() => setKind('binary')}>Binary source</Button>
      </div>
      <div data-toolbar-viewport className="h-[32rem] min-w-0 overflow-y-auto border">
        {kind === 'binary' ? (
          <FileDiffNotice oldPath="file.ts" newPath="file.ts" />
        ) : (
          <DiffView
            key={kind}
            ref={diff}
            model={model}
            defaultExpanded={false}
            annotationTargets={diffAnnotationTargets(model, kind)}
            oldPath="file.ts"
            newPath="file.ts"
          />
        )}
      </div>
    </>
  );
}
export default definePanel({
  id: 'diff-toolbar-acceptance',
  requires: { paneon: '>=0.0.1 <0.1.0' },
  input: z.object({}),
  render: Panel,
});
