import { FileDiffNotice } from './shared/file-diff-notice';
import { useState } from 'react';
import { definePanel, z } from 'paneon/sdk';
import { Button, AnnotationRoot, AnnotationMenu } from 'paneon/ui';
import { DiffView, diffTargetRanges, locateDiffTarget } from './shared/diff.tsx';
import { createDiffModel } from './shared/diff-model.ts';
import { diffTargetText } from './shared/diff-schema.ts';
import type { GitRef } from './shared/diff-schema.ts';

const original = Array.from(
  { length: 90 },
  (_, i) => `const line${i + 1} = "original ${i + 1}";\r\n`,
).join('');
const modified = original
  .replace('original 10', 'changed 10')
  .replace('original 45', 'changed 45')
  .replace('original 80', 'changed 80');
function Panel() {
  const [kind, setKind] = useState('changed');
  const model = createDiffModel(original, kind === 'same' ? original : modified);
  return (
    <AnnotationRoot<GitRef>
      onSubmit={async () => {}}
      adapter={{
        resolve: (target) => ({ status: 'ready', text: diffTargetText(model, target) }),
        targetRanges: (target) => diffTargetRanges(model, target),
        locate: (target) => locateDiffTarget(model, target),
      }}
    >
      <div className="flex flex-wrap items-center gap-2 p-2">
        <Button onClick={() => setKind('changed')}>Changed source</Button>
        <Button onClick={() => setKind('same')}>Same source</Button>
        <Button onClick={() => setKind('binary')}>Binary source</Button>
        <AnnotationMenu />
      </div>
      <div data-toolbar-viewport className="h-[32rem] min-w-0 overflow-y-auto border">
        {kind === 'binary' ? <FileDiffNotice oldPath="file.ts"
newPath="file.ts"/> : <DiffView
          sourceId={kind}
          oldPath="file.ts"
          newPath="file.ts"
          oldText={original}
          newText={kind === 'same' ? original : modified}

        />}
      </div>
    </AnnotationRoot>
  );
}
export default definePanel({
  id: 'diff-toolbar-acceptance',
  requires: { paneon: '0.0.0' },
  input: z.object({}),
  render: Panel,
});
