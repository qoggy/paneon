import { useMemo, useState } from 'react';
import { agent, definePanel, annotationValueSchema, z } from 'paneon/sdk';
import type { AnnotationValue } from 'paneon/sdk';
import {
  Button,
  AnnotationRoot,
  AnnotationMenu,
  AnnotationSubmitButton,
  AnnotationText,
  useAnnotations,
} from 'paneon/ui';
import type { AnnotationAdapter } from 'paneon/ui';

type Ref = string;
const text = 'Shared annotation target';
function Contents({ separate }: { separate: boolean }) {
  const feedback = useAnnotations<Ref>();
  const [extraMenu, setExtraMenu] = useState(false);
  return (
    <>
      <div className="flex flex-wrap items-center gap-2">
        {separate ? (
          <>
            <AnnotationMenu showSubmit={false} />
            <AnnotationSubmitButton />
          </>
        ) : (
          <AnnotationMenu />
        )}
        <Button variant="outline">Toolbar child</Button>
        <Button variant="outline" onClick={() => setExtraMenu(!extraMenu)}>
          Toggle extra menu
        </Button>
        {extraMenu && <AnnotationMenu showSubmit={false} />}
      </div>

      <p data-testid="target" className="mt-6 rounded-md border p-4">
        <AnnotationText target="shared">{text}</AnnotationText>
      </p>

      <output hidden data-testid="draft">
        {JSON.stringify(feedback.value)}
      </output>
    </>
  );
}
function Panel() {
  const [separate, setSeparate] = useState(false);
  const [disabled, setDisabled] = useState(false);
  const [located, setLocated] = useState('');
  const [locateCount, setLocateCount] = useState(0);
  const adapter = useMemo<AnnotationAdapter<Ref>>(
    () => ({
      resolve: () => ({ status: 'ready', text }),
      locate: (target) => {
        setLocated(target.ref);
        setLocateCount((count) => count + 1);
      },
    }),
    [],
  );
  return (
    <main className="space-y-4 p-4">
      <h1 className="text-lg font-semibold">Annotation editor acceptance</h1>
      <AnnotationRoot<Ref>
        disabled={disabled}
        adapter={adapter}
        onSubmit={async (value: AnnotationValue<Ref>) => {
          await agent.emit('feedback', value);
        }}
      >
        <Contents separate={separate} />
      </AnnotationRoot>
      <div className="flex flex-wrap gap-2" data-annotation-controls="">
        <Button variant="outline" onClick={() => setSeparate(!separate)}>
          Split toolbar {separate ? 'on' : 'off'}
        </Button>
        <Button variant="outline" onClick={() => setDisabled(!disabled)}>
          Disabled {disabled ? 'yes' : 'no'}
        </Button>
      </div>
      <output data-testid="located" data-locate-count={locateCount}>
        {located}
      </output>
    </main>
  );
}
export default definePanel({
  id: 'annotation-editor-acceptance',
  requires: { paneon: '0.0.0' },
  input: z.object({}),
  events: { feedback: annotationValueSchema(z.string()) },
  render: Panel,
});
