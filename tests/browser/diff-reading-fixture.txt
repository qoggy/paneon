import { useLayoutEffect, useMemo, useRef, useState } from 'react';
import { definePanel, z } from '@qoggy/paneon/sdk';
import { Button, useAnnotations, usePageAnnotations } from '@qoggy/paneon/ui';
import { DiffView } from './shared/diff.tsx';
import type { DiffViewHandle } from './shared/diff.tsx';
import { createDiffModel } from './shared/diff-model.ts';
import { diffAnnotationTargets, diffTargetLocation, diffTargetText } from './shared/diff-schema.ts';
import type { GitRef } from './shared/diff-schema.ts';
const original = Array.from(
  { length: 340 },
  (_, i) =>
    `const line${i + 1} = "original ${i + 1} ${'long_readable_source_'.repeat(i === 249 ? 60 : 1)}";\r\n`,
).join('');
const modified = original
  .replace('original 250', 'changed 250')
  .replace('original 290', 'changed 290');
const sources: Record<string, [string | undefined, string | undefined]> = {
  changed: [original, modified],
  same: [original, original],
  added: [undefined, modified],
  deleted: [original, undefined],
  filled: ['', modified],
  cleared: [original, ''],
  'spaces-old': [' \r\n', modified],
  'spaces-new': [original, ' \r\n'],
  empty: ['', ''],
  located: [original, modified],
};
function Draft() {
  const feedback = useAnnotations<GitRef>();
  return (
    <output data-reading-feedback className="sr-only">
      {JSON.stringify(feedback.value)}
    </output>
  );
}
function Panel() {
  const diff = useRef<DiffViewHandle>(null);
  const [kind, setKind] = useState('changed'),
    [collapsed, setCollapsed] = useState(false);
  const model = useMemo(() => createDiffModel(...sources[kind]), [kind]);
  const target = useMemo(
    () =>
      kind === 'located'
        ? {
            ref: {
              sourceId: kind,
              kind: 'lines' as const,
              side: 'new' as const,
              startLine: 12,
              endLine: 12,
            },
            text: { start: 0, end: 5, quote: 'const' },
          }
        : undefined,
    [kind],
  );
  useLayoutEffect(() => {
    if (!target) return;
    const frame = requestAnimationFrame(() => {
      void diff.current?.reveal(diffTargetLocation(model, target));
    });
    return () => cancelAnimationFrame(frame);
  }, [target, model]);
  usePageAnnotations<GitRef>({
    onSubmit: async () => {},
    adapter: {
      resolve: (target) => ({ status: 'ready', text: diffTargetText(model, target) }),
      targetRanges: (target) =>
        target.ref.sourceId === kind
          ? (diff.current?.getRanges(diffTargetLocation(model, target)) ?? [])
          : [],
      locate: (target) => diff.current?.reveal(diffTargetLocation(model, target)),
    },
  });
  return (
    <>
      <Draft />
      <div className="h-24 p-3">Diff reading acceptance</div>
      <div className="flex flex-wrap gap-1 p-2">
        {Object.keys(sources).map((value) => (
          <Button
            key={value}
            onClick={() => {
              setKind(value);
              setCollapsed(false);
            }}
          >
            {value}
          </Button>
        ))}
        <Button onClick={() => setCollapsed(false)}>Reopen</Button>
      </div>
      <div
        data-reading-kind={kind}
        data-reading-viewport
        className="h-[28rem] min-w-0 overflow-auto rounded-md border"
        hidden={collapsed}
        inert={collapsed}
      >
        <DiffView
          defaultExpanded
          key={kind}
          ref={diff}
          annotationTargets={diffAnnotationTargets(model, kind)}
          model={model}
          oldPath="packages/application/reader.ts"
          newPath="packages/application/reader.ts"
          oldRevision="764da2f4"
          newRevision="ba6b313e"
          onCollapse={() => setCollapsed(true)}
        />
      </div>
      <div className="h-[60rem]" />
    </>
  );
}
export default definePanel({
  id: 'diff-reading',
  requires: { paneon: '>=0.0.1 <0.1.0' },
  input: z.object({}),
  render: Panel,
});
