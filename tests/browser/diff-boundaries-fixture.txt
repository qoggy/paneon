import { useMemo, useRef } from 'react';
import { definePanel, z } from 'paneon/sdk';
import { AnnotationRoot, AnnotationMenu } from 'paneon/ui';
import type { GitRef } from './shared/diff-schema.ts';
import { DiffView } from './shared/diff.tsx';
import type { DiffViewHandle } from './shared/diff.tsx';
import { createDiffModel } from './shared/diff-model.ts';
import { diffAnnotationTargets, diffTargetLocation, diffTargetText } from './shared/diff-schema.ts';
const original = Array.from({ length: 140 }, (_, i) => `source ${i + 1} 中文é🙂\r\n`).join('');
const insertion = Array.from({ length: 82 }, (_, i) => `insert ${i + 1} 中文é🙂\r\n`).join('');
const modified = original
  .replace('source 29', `${insertion}source 29`)
  .replace('source 33 中文é🙂\r\nsource 34', 'updated 33 中文é🙂\r\nupdated 34');
function Panel() {
  const diff = useRef<DiffViewHandle>(null);
  const model = useMemo(() => createDiffModel(original, modified), []);
  return (
    <AnnotationRoot<GitRef>
      onSubmit={async () => {}}
      adapter={{
        resolve: (target) => ({ status: 'ready', text: diffTargetText(model, target) }),
        targetRanges: (target) => diff.current?.getRanges(diffTargetLocation(model, target)) ?? [],
        locate: (target) => diff.current?.reveal(diffTargetLocation(model, target)),
      }}
    >
      <div className="h-32 p-4">History outside the diff viewport</div>
      <AnnotationMenu />
      <div data-boundary-viewport className="h-[28rem] overflow-auto border">
        <DiffView
          ref={diff}
          annotationTargets={diffAnnotationTargets(model, 'merged-context')}
          model={model}
          oldPath="file.md"
          newPath="file.md"
          defaultExpanded
        />
      </div>
      <div className="h-96">Page outside the diff viewport</div>
    </AnnotationRoot>
  );
}
export default definePanel({
  id: 'diff-boundaries',
  requires: { paneon: '0.0.0' },
  input: z.object({}),
  render: Panel,
});
