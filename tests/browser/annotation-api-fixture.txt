import { agent, definePanel, annotationValueSchema, z } from '@qoggy/paneon/sdk';
import {
  Page,
  Card,
  AnnotationText,
  toast,
  useAnnotations,
  usePageAnnotations,
} from '@qoggy/paneon/ui';

const source = 'Use Redis for shared cache storage.';
const refSchema = z.union([
  z.literal('redis-plan'),
  z.object({
    id: z.object({ page: z.literal('example'), version: z.literal(1) }),
    field: z.enum(['title', 'description']),
  }),
]);
type Ref = z.infer<typeof refSchema>;
const draftAnnotation = annotationValueSchema(refSchema).shape.annotations.element;
const resultSchema = z.object({
  annotations: z.array(
    z.object({
      targets: draftAnnotation.shape.targets,
      comment: draftAnnotation.shape.comment,
    }),
  ),
});

function Draft() {
  const { value } = useAnnotations<Ref>();
  return (
    <output data-draft hidden>
      {JSON.stringify(value)}
    </output>
  );
}

function Panel() {
  usePageAnnotations<Ref>({
    onSubmit: async (value) => {
      const receipt = await agent.emit('annotations-submitted', {
        annotations: value.annotations.map(({ targets, comment }) => ({ targets, comment })),
      });
      toast.add({ title: `Saved as event ${receipt.id}.`, type: 'success' });
    },
  });
  return (
    <Page
      title="Annotation example"
      description="Annotation description"
      annotationTarget={{ page: 'example', version: 1 }}
    >
      <>
        <Card className="p-4">
          <AnnotationText target="redis-plan">{source}</AnnotationText>
        </Card>
        <Draft />
      </>
    </Page>
  );
}

export default definePanel({
  id: 'custom-annotations',
  requires: { paneon: '__PANEON_TEST_VERSION__' },
  input: z.object({}).strict(),
  events: { 'annotations-submitted': resultSchema },
  render: Panel,
});
